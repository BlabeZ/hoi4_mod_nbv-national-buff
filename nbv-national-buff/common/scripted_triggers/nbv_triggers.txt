        # 544: encryption_factor; P2; runtime_read_validated=true
        if = {
            limit = { OR = { has_country_flag = nbv_show_zero NOT = { check_variable = { ROOT.modifier@encryption_factor = 0 } } } }
            add_to_temp_array = { nbv_low_entries = 544 }
        }
        # 542: decryption_factor; P2; runtime_read_validated=true
        if = {
            limit = { OR = { has_country_flag = nbv_show_zero NOT = { check_variable = { ROOT.modifier@decryption_factor = 0 } } } }
            add_to_temp_array = { nbv_low_entries = 542 }
        }
    }
    set_temp_variable = { nbv_high_count = nbv_high_entries^num }
    set_temp_variable = { nbv_low_count = nbv_low_entries^num }
    nbv_prepare_layout = yes
    always = yes
}
